\newcommand{\bibliografiateste}[1]{%
  \defbibheading{undfreferences}{\section*{BIBLIOGRAFIA-CASO-#1}}%
  \printundfreferences
  \par FIM-CASO-#1\par
}

\parenciteapud{Origem}{1980}[3]{consultada}[9].
\textapud{Origem}{1980}{consultada}.
\bibliografiateste{1}
\end{refsection}

\end{refsection}
